	public static int[] meshalka(int[] arr2)// Сортировка перемешивания 
	{
		if (arr2 == null || arr2.Length == 0)
		{
			return arr2;
		}
		int left = 0;
		int right = arr2.Length - 1;

		while (left < right)
		{
			for (int i = left; i < arr2.Length - 1; i++)
			{
				if (arr2[i] > arr2[i + 1])
				{
					int indicator = arr2[i];
					arr2[i] = arr2[i + 1];
					arr2[i + 1] = indicator;

				}
			}
			right--;

			for (int i = right; i >= 1; i--)
			{
				if (arr2[i - 1] > arr2[i])
				{
					int indicator = arr2[i];
					arr2[i] = arr2[i - 1];
					arr2[i - 1] = indicator;

				}
			}
			left++;
		}

		return arr2;
	}

	public static int[] dwarfSorting(int[] arr1)// гномья сортировка 
	{
		if (arr1 == null || arr1.Length == 0)
		{
			return arr1;
		}

		int indicator = 1;
		while (indicator < arr1.Length)
		{

			if (indicator == 0 || arr1[indicator] >= arr1[indicator - 1])
			{
				indicator++;
			}
			else
			{
				int dopIndicator = arr1[indicator];
				arr1[indicator] = arr1[indicator - 1];
				arr1[indicator - 1] = dopIndicator;

				indicator--;
			}
		}
		return arr1;

	}

	public static double function(double a, double b)// метод по вычислению функции
	{
		double grad;
		const double PI = Math.PI;
		grad = b * PI / 180;
		double f = Math.Pow(Math.Sin(3 * PI / 4 + a / 3), 2) + Math.Sqrt(grad);
		return f;
	}
